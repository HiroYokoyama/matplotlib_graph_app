    def create_scrollable_tab(self, parent_notebook, tab_name, create_content_func):
        tab_container = ttk.Frame(parent_notebook)
        parent_notebook.add(tab_container, text=tab_name)

        tab_canvas = tk.Canvas(tab_container, borderwidth=0, highlightthickness=0)
        tab_scrollbar = ttk.Scrollbar(tab_container, orient=tk.VERTICAL, command=tab_canvas.yview)
        tab_canvas.configure(yscrollcommand=tab_scrollbar.set)

        tab_scrollbar.pack(side=tk.RIGHT, fill=tk.Y)
        tab_canvas.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)

        content_frame = ttk.Frame(tab_canvas, padding=5)
        canvas_window = tab_canvas.create_window((0, 0), window=content_frame, anchor="nw")

        def on_content_configure(event):
            tab_canvas.configure(scrollregion=tab_canvas.bbox("all"))

        content_frame.bind("<Configure>", on_content_configure)

        def _scroll_cb(scroll_units, event):
            tab_canvas.yview_scroll(scroll_units, "units")

        bind_scroll_events(tab_canvas, _scroll_cb)
        create_content_func(content_frame)
        bind_mousewheel_recursive(content_frame, _scroll_cb)
        return content_frame

    def load_data(self, file_path=None):
        if file_path is None:
            file_path = filedialog.askopenfilename(
                title="Select Data File",
                filetypes=[("CSV files", "*.csv"), ("Excel files", "*.xlsx *.xls"), ("All files", "*.*")]
            )
        if not file_path:
            return

        try:
            self.data_mgr.load_file(file_path)
        except Exception as e:
            messagebox.showerror("Load Error", f"Failed to read file:\n{e}")
            return

        if self.sheet:
            self.sheet.destroy()

        self.sheet = Sheet(
            self.sheet_frame,
            data=self.df.values.tolist(),
            headers=self.df.columns.tolist(),
            show_toolbar=True,
            show_top_left=True
        )
        self.sheet.enable_bindings()
        self.sheet.pack(fill=tk.BOTH, expand=True)

        self.update_plot_options()

    def update_plot_options(self):
        if self.df is None:
            return

        cols = self.data_mgr.get_columns()
        self.x_axis_combo['values'] = cols
        self.x_axis_combo['state'] = 'readonly'
        self.y_axis_combo['values'] = cols
        self.y_axis_combo['state'] = 'readonly'

        self.y_listbox.delete(0, tk.END)
        for c in cols:
            self.y_listbox.insert(tk.END, c)

        self.plot_button['state'] = 'normal'
        self.export_button['state'] = 'normal'

        if cols:
            self.x_axis_var.set(cols[0])
            if len(cols) > 1:
                self.y_axis_var.set(cols[1])
            if len(cols) > 2:
                self.y_listbox.select_set(2)

    def get_data_from_sheet(self):
        if not self.sheet or self.df is None:
            return
        try:
            data = self.sheet.get_sheet_data() if hasattr(self.sheet, 'get_sheet_data') else getattr(self.sheet, 'data', None)
            headers = self.sheet.get_headers() if hasattr(self.sheet, 'get_headers') else getattr(self.sheet, 'headers', None)
            if callable(headers):
                headers = headers()
            self.data_mgr.update_from_sheet_data(data, headers)
        except Exception as e:
            print(f"Sheet retrieval error: {e}")

    def plot_graph(self):
        try:
            self.fig.clear()
            self.ax = self.fig.add_subplot(111, projection='3d')
            self.canvas.draw()
        except Exception as e:
            messagebox.showerror("Internal Error", f"Failed to clear 3D graph:\n{e}")
            return

        self.get_data_from_sheet()
        if not self.data_mgr.has_data():
            messagebox.showinfo("Info", "No data to plot.")
            return

        x_col = self.x_axis_var.get()
        y_col = self.y_axis_var.get()
        z_cols = [self.y_listbox.get(i) for i in self.y_listbox.curselection()]

        if not x_col or not y_col or not z_cols:
            messagebox.showerror("Error", "Please select X, Y, and at least one Z column.")
            return

        try:
            self.ax.view_init(elev=self.view_elev_var.get(), azim=self.view_azim_var.get())

            x_num = pd.to_numeric(self.df[x_col].astype(str).str.replace(r'[^\d.-]', '', regex=True), errors='coerce')
            y_num = pd.to_numeric(self.df[y_col].astype(str).str.replace(r'[^\d.-]', '', regex=True), errors='coerce')

            plot_type = self.plot_type_var.get()

            for z_c in z_cols:
                z_num = pd.to_numeric(self.df[z_c].astype(str).str.replace(r'[^\d.-]', '', regex=True), errors='coerce')
                valid_df = pd.DataFrame({'x': x_num, 'y': y_num, 'z': z_num}).dropna()

                if valid_df.empty:
                    continue

                if plot_type in ["surface", "wireframe", "contour3d"]:
                    res = self.mesh_resolution_var.get()
                    xi = np.linspace(valid_df['x'].min(), valid_df['x'].max(), res)
                    yi = np.linspace(valid_df['y'].min(), valid_df['y'].max(), res)
                    Xi, Yi = np.meshgrid(xi, yi)
                    Zi = griddata((valid_df['x'].values, valid_df['y'].values), valid_df['z'].values, (Xi, Yi), method='linear')
                    Zi = np.nan_to_num(Zi, nan=np.nanmean(valid_df['z'].values))

                    if plot_type == "surface":
                        self.ax.plot_surface(Xi, Yi, Zi, cmap=self.colormap_var.get(), alpha=0.8)
                    elif plot_type == "wireframe":
                        self.ax.plot_wireframe(Xi, Yi, Zi, rstride=5, cstride=5)
                    elif plot_type == "contour3d":
                        self.ax.contour3D(Xi, Yi, Zi, 20, cmap=self.colormap_var.get())

                elif plot_type == "scatter3d":
                    self.ax.scatter(valid_df['x'], valid_df['y'], valid_df['z'], label=z_c)
                elif plot_type == "line3d":
                    self.ax.plot(valid_df['x'], valid_df['y'], valid_df['z'], label=z_c)

            self.ax.set_xlabel(x_col)
            self.ax.set_ylabel(y_col)
            self.ax.set_zlabel(", ".join(z_cols))
            if self.title_var.get():
                self.ax.set_title(self.title_var.get())

            self.canvas.draw()
        except Exception as e:
            messagebox.showerror("Plot Error", f"Failed to plot 3D graph:\n{e}")

        messagebox.showinfo("Success", f"Project saved to {self.current_project_path}.")

    def reset_settings(self):
        self.plot_type_var.set("surface")
        self.view_elev_var.set(30)
        self.view_azim_var.set(-60)

    def setup_drag_and_drop(self):
        if DND_AVAILABLE:
            try:
                from tkinterdnd2 import DND_FILES
                self.drop_target_register(DND_FILES)
                self.dnd_bind('<<Drop>>', self.on_drop)
            except Exception as e:
                print(f"DnD error: {e}")

    def on_drop(self, event):
        files = getattr(event, 'data', '')
        if files:
            file_path = str(files).strip('{}').strip()
            ext = os.path.splitext(file_path)[1].lower()
            if ext == '.pmggrp':
                self.load_project_file(file_path)
            elif ext in ['.csv', '.xlsx', '.xls']:
                self.load_data(file_path=file_path)

    def load_project_file(self, file_path):
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                settings = json.load(f)
            if 'edited_data' in settings and settings['edited_data']:
                d = settings['edited_data']
                self.df = pd.DataFrame(d['data'], columns=d['columns']).astype(str).fillna("")
                if self.sheet:
                    self.sheet.destroy()
                self.sheet = Sheet(self.sheet_frame, data=self.df.values.tolist(), headers=self.df.columns.tolist())
                self.sheet.enable_bindings()
                self.sheet.pack(fill=tk.BOTH, expand=True)
                self.update_plot_options()

            self.current_project_path = file_path
            self.overwrite_save_button['state'] = 'normal'
            if self.df is not None:
                self.plot_graph()
        except Exception as e:
            messagebox.showerror("Error", f"Failed to load project file:\n{e}")

    def open_in_2d_mode(self):
        import subprocess
        script_dir = os.path.dirname(os.path.abspath(__file__))
        main_path = os.path.join(script_dir, "main.py")
        cmd = [sys.executable, main_path]
        if self.current_project_path and os.path.exists(self.current_project_path):
            cmd.append(self.current_project_path)
        subprocess.Popen(cmd)
